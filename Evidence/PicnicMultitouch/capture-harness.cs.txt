using System;
using System.IO;
using System.Reflection;
using UnityEditor.SceneManagement;
using UnityEngine;
using YunaLegend.Picnic.Core;
using YunaLegend.Picnic.Runtime;
namespace YunaLegend.EditorScripts
{
    public static class PicnicMultitouchCapture
    {
        public static void CheckOriginal()
        {
            typeof(RoundedHudReviewCapture).GetMethod("CapturePicnic", BindingFlags.NonPublic | BindingFlags.Static).Invoke(null, null);
        }

        public static void Capture()
        {
            EditorSceneManager.OpenScene("Assets/Scenes/Picnic_Gameplay.unity");
            var camera = UnityEngine.Object.FindFirstObjectByType<Camera>();
            var hud = UnityEngine.Object.FindFirstObjectByType<PicnicNarrativeAndHudController>();
            var platforms = UnityEngine.Object.FindFirstObjectByType<PicnicPlatformController>();
            var scenery = UnityEngine.Object.FindFirstObjectByType<PicnicSceneryController>();
            var actors = UnityEngine.Object.FindFirstObjectByType<PicnicActorController>();
            var sim = new PicnicSimulation();
            platforms.Build(sim.Level); scenery.Build(camera, sim.Level); actors.Build(sim); hud.Build();
            sim.SkipStory(); platforms.Apply(sim); actors.Apply(sim); scenery.Apply(sim); hud.Apply(sim, false);
            Directory.CreateDirectory("Logs/mobile_ui_audit");
            var render = typeof(MobileUiCapture).GetMethod("RenderSceneWithUi", BindingFlags.NonPublic | BindingFlags.Static);
            foreach (var size in new [] { new Vector2Int(320,568), new Vector2Int(390,844), new Vector2Int(844,390) })
            {
                render.Invoke(null, new object[] { camera, size.x, size.y, (Action)(() => {
                    hud.SetViewportOverride(size.x,size.y); Canvas.ForceUpdateCanvases(); hud.SetViewportOverride(size.x,size.y); Canvas.ForceUpdateCanvases();
                }), "multitouch-"+size.x+"x"+size.y+".png" });
            }
            Debug.Log("PICNIC_MULTITOUCH_CAPTURE_OK");
        }
    }
}
