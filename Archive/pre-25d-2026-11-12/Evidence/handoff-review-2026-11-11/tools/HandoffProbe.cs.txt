using System;
using System.IO;
using System.Linq;
using System.Collections.Generic;
using UnityEditor;
using UnityEditor.SceneManagement;
using UnityEngine;
using YunaLegend.DarkRoom;

public static class HandoffProbe
{
    [Serializable] class Report { public Platform[] platforms; public Gap[] gaps; }
    [Serializable] class Platform
    {
        public int index, samples, missingVisual, visualBelow, visualAbove, aligned, renderers, triangles, materials, z0Missing, z0Below, z0Above;
        public float colliderWidth, colliderDepth, minGap, maxGap, centerGap, z0MaxGap;
        public Vector3 colliderTop, visualMin, visualMax, headBounds, coverBounds;
        public Vector3[] colliderCorners, coverCorners;
        public int[] support = new int[441];
    }
    [Serializable] class Gap { public int from, to; public float volumeGap, contactGap; public bool generatorDeclaresCustomContact; }
    public static void Run()
    {
        string output = Environment.GetEnvironmentVariable("YUNA_HANDOFF_AUDIT_OUTPUT");
        if (string.IsNullOrEmpty(output)) throw new Exception("Set YUNA_HANDOFF_AUDIT_OUTPUT");
        Directory.CreateDirectory(output);
        EditorSceneManager.OpenScene(Environment.GetEnvironmentVariable("YUNA_HANDOFF_SCENE") ?? "Assets/Scenes/DarkRoom_Gameplay.unity");
        var markers = UnityEngine.Object.FindObjectsByType<DarkRoomPlatformMarker>(FindObjectsInactive.Include, FindObjectsSortMode.None);
        var report = new Report { platforms = new[] { Probe(markers.Single(m => m.spec.index == 0)), Probe(markers.Single(m => m.spec.index == 1)) } };
        var stage = DarkRoomThreeStageCircuitGenerator.GenerateStage1Platforms();
        var gaps = new List<Gap>();
        for (int i = 0; i < stage.Count - 1; i++)
        {
            var a = markers.Single(m => m.spec.index == i); var b = markers.Single(m => m.spec.index == i+1);
            float[] ea = ContactEdges(a), eb = ContactEdges(b);
            gaps.Add(new Gap { from=i, to=i+1,
                volumeGap=b.spec.x-b.spec.width*.5f-(a.spec.x+a.spec.width*.5f),
                contactGap=eb[0]-ea[1], generatorDeclaresCustomContact=stage[i+1].contact.IsCustom });
        }
        report.gaps=gaps.ToArray();
        File.WriteAllText(Path.Combine(output,"probe.json"),JsonUtility.ToJson(report,true));
        Debug.Log("HANDOFF_PROBE_COMPLETE " + output);
    }
    static float[] ContactEdges(DarkRoomPlatformMarker marker)
    {
        float half=marker.spec.contact.EffectiveWidth(marker.spec.width)*.5f;
        var center=marker.GetTopCenterWorld(); var rotation=marker.GetContactRotation();
        float x1=(center+rotation*Vector3.right*half).x, x2=(center-rotation*Vector3.right*half).x;
        return new[]{Mathf.Min(x1,x2),Mathf.Max(x1,x2)};
    }
    static Vector3[] Corners(Transform t, Vector3 center, Vector3 size)
    {
        var points=new List<Vector3>();
        foreach(int x in new[]{-1,1}) foreach(int z in new[]{-1,1}) points.Add(t.TransformPoint(center+new Vector3(x*size.x*.5f,size.y*.5f,z*size.z*.5f)));
        return points.ToArray();
    }
    static Platform Probe(DarkRoomPlatformMarker marker)
    {
        var box=marker.transform.Find("Contacto_Fisico").GetComponent<BoxCollider>();
        var renderers=marker.transform.Find("Visual_3D").GetComponentsInChildren<MeshRenderer>();
        var filters=marker.transform.Find("Visual_3D").GetComponentsInChildren<MeshFilter>();
        var meshColliders=new List<MeshCollider>();
        foreach(var f in filters) { var c=f.gameObject.AddComponent<MeshCollider>(); c.sharedMesh=f.sharedMesh; meshColliders.Add(c); }
        Physics.SyncTransforms();
        var bounds=renderers[0].bounds;foreach(var r in renderers) bounds.Encapsulate(r.bounds);
        var p=new Platform { index=marker.spec.index,colliderWidth=box.size.x,colliderDepth=box.size.z,
            colliderTop=box.transform.TransformPoint(box.center+Vector3.up*box.size.y*.5f),
            colliderCorners=Corners(box.transform,box.center,box.size),visualMin=bounds.min,visualMax=bounds.max,
            renderers=renderers.Length,triangles=filters.Sum(f=>f.sharedMesh.triangles.Length/3),
            materials=renderers.SelectMany(r=>r.sharedMaterials).Where(m=>m!=null).Distinct().Count(), minGap=float.MaxValue,maxGap=float.MinValue };
        var head=marker.transform.Find("Visual_3D/Cabeca_Urso");
        if(head!=null) p.headBounds=head.GetComponent<MeshRenderer>().bounds.size;
        var cover=marker.transform.Find("Visual_3D/Livro_05/Capa_Superior");
        if(cover!=null) { p.coverBounds=cover.GetComponent<MeshRenderer>().bounds.size;p.coverCorners=Corners(cover,Vector3.zero,Vector3.one); }
        for(int xi=0;xi<=20;xi++) for(int zi=0;zi<=20;zi++)
        {
            // Interior grid avoids ambiguous triangles exactly at a boundary.
            var local=box.center+new Vector3((xi/20f-.5f)*box.size.x*.998f,box.size.y*.5f,(zi/20f-.5f)*box.size.z*.998f);
            Vector3 point=box.transform.TransformPoint(local), normal=box.transform.up;
            var ray=new Ray(point+normal*3f,-normal);float nearest=float.MaxValue;
            foreach(var c in meshColliders) if(c.Raycast(ray,out var hit,20f)) nearest=Mathf.Min(nearest,hit.distance);
            p.samples++;
            if(nearest==float.MaxValue) {p.missingVisual++;if(zi==10)p.z0Missing++;continue;}
            float gap=nearest-3f;
            p.minGap=Mathf.Min(p.minGap,gap);p.maxGap=Mathf.Max(p.maxGap,gap);
            if(Mathf.Abs(gap)<=.02f) {p.aligned++;p.support[xi*21+zi]=1;} else if(gap>.02f) {p.visualBelow++;p.support[xi*21+zi]=2;} else {p.visualAbove++;p.support[xi*21+zi]=3;}
            if(zi==10) {p.z0MaxGap=Mathf.Max(p.z0MaxGap,gap);if(gap>.02f)p.z0Below++;if(gap<-.02f)p.z0Above++;}
            if(xi==10&&zi==10)p.centerGap=gap;
        }
        foreach(var c in meshColliders)UnityEngine.Object.DestroyImmediate(c);
        return p;
    }
}
