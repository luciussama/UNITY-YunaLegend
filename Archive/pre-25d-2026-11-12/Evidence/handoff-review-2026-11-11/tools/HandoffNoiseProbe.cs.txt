using System;
using System.IO;
using System.Linq;
using System.Security.Cryptography;
using System.Collections.Generic;
using UnityEngine;
using YunaLegend.EditorScripts;

public static class HandoffNoiseProbe
{
    [Serializable] class Report { public string[] changedBetweenRegenerations; public int compared; }
    static Dictionary<string,string> Hashes()
    {
        using(var sha=SHA256.Create()) return Directory.GetFiles("Assets/Audio/SFX","*.wav").ToDictionary(
            p=>Path.GetFileName(p),p=>BitConverter.ToString(sha.ComputeHash(File.ReadAllBytes(p))));
    }
    public static void Run()
    {
        // Run only inside an isolated validation project; this intentionally rewrites its WAVs.
        DarkRoomAudioAssetBuilder.GenerateAllSfx();var first=Hashes();
        System.Threading.Thread.Sleep(50);
        DarkRoomAudioAssetBuilder.GenerateAllSfx();var second=Hashes();
        var report=new Report {compared=first.Count,changedBetweenRegenerations=first.Keys.Where(k=>first[k]!=second[k]).OrderBy(k=>k).ToArray()};
        string output=Environment.GetEnvironmentVariable("YUNA_HANDOFF_AUDIT_OUTPUT");
        File.WriteAllText(Path.Combine(output,"audio-regeneration.json"),JsonUtility.ToJson(report,true));
        HandoffProbe.Run();
        Debug.Log("HANDOFF_NOISE_PROBE_COMPLETE changed="+report.changedBetweenRegenerations.Length);
    }
}
